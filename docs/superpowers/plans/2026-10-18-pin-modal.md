# 10,000-Pin Modal Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Press `P` on the audience screen to open a modal showing the current section as a 3D bed of
100×100 coloured pins (drag to turn, pins slide between sections, a "pins per side" slider down to
today's 10×10 board), and show the same modal in the Paper design file.

**Architecture:** The backend already serves `section.relief` and `section.relief_color` (100×100 PNG
URLs, see `docs/relief.md`). A pure module `src/lib/pins.ts` turns those pixels into a `PinField`
(heights + colours) and resamples it; `src/lib/pinScene.ts` owns all three.js code (one
`InstancedMesh`, `OrbitControls`, the per-frame height animation); `src/components/PinModal.tsx` loads
the images, holds the slider state and hosts the scene. `App.tsx` toggles it with `P`, like `S` toggles
the drawer.

**Tech Stack:** React 18 + Vite 5 + TypeScript 5.6, three.js `0.170.0` (+ `@types/three` `0.170.0`),
node:test on Node 24 (type stripping, existing pattern), Paper MCP, Playwright MCP for screenshots.

**Spec:** the short design agreed in chat on 2026-10-18 (bounded path, no spec file). Summary: `P`
toggles a modal over the screen; 100×100 pins coloured like the painting; drag to orbit, scroll to zoom;
pins slide to new heights when the section changes; one resolution slider (10 / 20 / 25 / 50 / 100 pins
per side), where 10 shows today's board; caption labels it a simulation; `P` or `Esc` closes; a
"making the relief" message while `relief` is `null`.

```ts
import { test } from "node:test";
import assert from "node:assert/strict";
import { boardField, fieldFromPixels, resample, RESOLUTIONS, type Pixels } from "../src/lib/pins.ts";

// size x size RGBA image where pixel (x, y) = fn(x, y) -> [r, g, b]
function img(size: number, fn: (x: number, y: number) => [number, number, number]): Pixels {
  const data = new Uint8ClampedArray(size * size * 4);
  for (let y = 0; y < size; y++)
    for (let x = 0; x < size; x++) {
      const [r, g, b] = fn(x, y);
      data.set([r, g, b, 255], (y * size + x) * 4);
    }
  return { width: size, height: size, data };
}

test("heights come from the red channel, colours from the colour image, row-major", () => {
  const h = img(4, (x, y) => [x === 1 && y === 2 ? 255 : 0, 0, 0]);
  const c = img(4, () => [255, 0, 51]);
  const f = fieldFromPixels(h, c);
  assert.equal(f.size, 4);
  assert.equal(f.heights[2 * 4 + 1], 1);
  assert.equal(f.heights[0], 0);
  assert.deepEqual([...f.colors.slice(0, 2)], [1, 0]);
  assert.ok(Math.abs(f.colors[2] - 0.2) < 1e-6); // Float32 can't hold 0.2 exactly
});

test("resample averages blocks of heights and colours", () => {
  // left half 255, right half 0 -> at 2x2 each left cell is 1, right 0; at 1x1 it's 0.5
  const h = img(4, (x) => [x < 2 ? 255 : 0, 0, 0]);
  const f = fieldFromPixels(h, h);
  const two = resample(f, 2);
  assert.deepEqual([...two.heights], [1, 0, 1, 0]);
  const one = resample(f, 1);
  assert.equal(one.size, 1);
  assert.equal(one.heights[0], 0.5);
  assert.equal(one.colors[0], 0.5);
  assert.equal(resample(f, 4), f, "same size returns the field unchanged");
});

test("boardField takes heights from the real 10x10 grid and keeps the colours", () => {
  const f = resample(fieldFromPixels(img(100, () => [128, 10, 20]), img(100, () => [255, 0, 0])), 10);
  const grid = Array.from({ length: 10 }, (_, r) => Array.from({ length: 10 }, (_, c) => (r === 0 && c === 9 ? 1 : 0)));
  const b = boardField(f, grid);
  assert.equal(b.heights[9], 1);
  assert.equal(b.heights[0], 0);
  assert.equal(b.colors[0], 1);
});

test("every resolution divides 100", () => {
  assert.deepEqual([...RESOLUTIONS], [10, 20, 25, 50, 100]);
  assert.ok(RESOLUTIONS.every((n) => 100 % n === 0));
```

```ts
// The 10,000-pin simulation's data (docs/relief.md), kept free of three.js and the DOM so node can test it.

export interface Pixels { width: number; height: number; data: Uint8ClampedArray }

// heights[r * size + c] in 0..1; colors[(r * size + c) * 3 + k] = sRGB channel k in 0..1. Row 0 = far edge.
export interface PinField { size: number; heights: Float32Array; colors: Float32Array }

export const RESOLUTIONS: readonly number[] = [10, 20, 25, 50, 100];

// Pin height = red channel of the relief; colour = the relief_color pixel under it.
export function fieldFromPixels(height: Pixels, color: Pixels): PinField {
  const size = height.width;
  const heights = new Float32Array(size * size);
  const colors = new Float32Array(size * size * 3);
  for (let i = 0; i < size * size; i++) {
    heights[i] = height.data[i * 4] / 255;
    for (let k = 0; k < 3; k++) colors[i * 3 + k] = color.data[i * 4 + k] / 255;
  }
  return { size, heights, colors };
}

// n x n pins, each the average of a (size/n) x (size/n) block. n must divide size.
export function resample(f: PinField, n: number): PinField {
  if (n === f.size) return f;
  const b = f.size / n;
  const heights = new Float32Array(n * n);
  const colors = new Float32Array(n * n * 3);
  for (let r = 0; r < n; r++)
    for (let c = 0; c < n; c++) {
      let h = 0;
      const rgb = [0, 0, 0];
      for (let y = r * b; y < (r + 1) * b; y++)
        for (let x = c * b; x < (c + 1) * b; x++) {
          const i = y * f.size + x;
          h += f.heights[i];
          for (let k = 0; k < 3; k++) rgb[k] += f.colors[i * 3 + k];
        }
      const o = r * n + c;
      heights[o] = h / (b * b);
      for (let k = 0; k < 3; k++) colors[o * 3 + k] = rgb[k] / (b * b);
    }
  return { size: n, heights, colors };
}

// Today's board: the section's real up/down grid, coloured like the painting. f must be 10x10.
export function boardField(f: PinField, grid: number[][]): PinField {
  const heights = new Float32Array(100);
  for (let r = 0; r < 10; r++) for (let c = 0; c < 10; c++) heights[r * 10 + c] = grid[r]?.[c] ? 1 : 0;
  return { size: 10, heights, colors: f.colors };
}
```

```ts
// All the three.js for the 10,000-pin bed: one InstancedMesh (one draw call for every pin), orbit controls,
// and pins that slide toward their new heights each frame.
import * as THREE from "three";
import { OrbitControls } from "three/examples/jsm/controls/OrbitControls.js";
import { reducedMotion } from "./motion";
import type { PinField } from "./pins";

const BED = 10;        // world units across the bed, whatever the resolution
const MAX = 100 * 100; // instance capacity
const STUB = 0.06;     // a pin at height 0 still shows as a short stub
const RISE = 2.2;      // world height of a fully raised pin
const EASE = 0.12;     // share of the remaining distance a pin moves per frame

export interface PinScene { show(field: PinField): void; dispose(): void }

export function createPinScene(host: HTMLElement): PinScene {
  const still = reducedMotion();
  const renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  host.appendChild(renderer.domElement);

  const scene = new THREE.Scene();
  scene.background = new THREE.Color("#08090b");
  const camera = new THREE.PerspectiveCamera(35, 1, 0.1, 100);
  camera.position.set(0, 10, 12);
  scene.add(new THREE.AmbientLight("#ffffff", 0.7));
  const sun = new THREE.DirectionalLight("#fff1dc", 1.8);
  sun.position.set(4, 12, 6);
  scene.add(sun);

  const controls = new OrbitControls(camera, renderer.domElement);
  controls.target.set(0, 0.6, 0);
  controls.enableDamping = true;
  controls.enablePan = false;
  controls.minDistance = 6;
  controls.maxDistance = 26;
  controls.maxPolarAngle = Math.PI / 2.1; // never look from under the bed
  controls.autoRotate = !still;
  controls.autoRotateSpeed = 0.6;
  controls.addEventListener("start", () => { controls.autoRotate = false; }); // a visitor took over

  const geometry = new THREE.BoxGeometry(1, 1, 1);
  geometry.translate(0, 0.5, 0); // scale.y grows the pin up from the bed, not both ways
  const material = new THREE.MeshStandardMaterial({ roughness: 0.55, metalness: 0.1 });
  const mesh = new THREE.InstancedMesh(geometry, material, MAX);
  mesh.count = 0;
  scene.add(mesh);

  let size = 0;
  let target = new Float32Array(0);
  let current = new Float32Array(0);
  const m = new THREE.Matrix4();
  const color = new THREE.Color();

  function place() {
    const pitch = BED / size;
    const w = pitch * 0.82;
    for (let r = 0; r < size; r++)
      for (let c = 0; c < size; c++) {
        const i = r * size + c;
        // row 0 at -z (far from the visitor), column 0 at -x (visitor's left)
        m.makeScale(w, STUB + current[i] * RISE, w);
        m.setPosition((c - (size - 1) / 2) * pitch, 0, (r - (size - 1) / 2) * pitch);
        mesh.setMatrixAt(i, m);
      }
    mesh.instanceMatrix.needsUpdate = true;
  }

  function show(field: PinField) {
    if (field.size !== size) {
      size = field.size;
      current = new Float32Array(size * size); // a new resolution rises from flat
      mesh.count = size * size;
    }
    target = field.heights;
    if (still) current = Float32Array.from(target);
    for (let i = 0; i < size * size; i++) {
      color.setRGB(field.colors[i * 3], field.colors[i * 3 + 1], field.colors[i * 3 + 2], THREE.SRGBColorSpace);
      mesh.setColorAt(i, color);
    }
    if (mesh.instanceColor) mesh.instanceColor.needsUpdate = true;
    place();
  }

  function resize() {
    const w = host.clientWidth || 1;
    const h = host.clientHeight || 1;
    renderer.setSize(w, h);
    camera.aspect = w / h;
    camera.updateProjectionMatrix();
  }
  const observer = new ResizeObserver(resize);
  observer.observe(host);
  resize();

  let frame = 0;
  function tick() {
    frame = requestAnimationFrame(tick);
    let moving = false;
    for (let i = 0; i < size * size; i++) {
      const d = target[i] - current[i];
      if (Math.abs(d) > 0.001) { current[i] += d * EASE; moving = true; }
    }
    if (moving) place();
    controls.update();
    renderer.render(scene, camera);
  }
  tick();

  return {
    show,
    dispose() {
      cancelAnimationFrame(frame);
      observer.disconnect();
      controls.dispose();
      geometry.dispose();
      material.dispose();
      mesh.dispose();
      renderer.dispose();
      renderer.domElement.remove();
    },
  };
}
```

```tsx
import { useEffect, useMemo, useRef, useState } from "react";
import { COPY } from "../design/copy";
import { createPinScene, type PinScene } from "../lib/pinScene";
import { boardField, fieldFromPixels, resample, RESOLUTIONS, type PinField, type Pixels } from "../lib/pins";
import type { Section } from "../types";
import "./PinModal.css";

function loadPixels(url: string): Promise<Pixels> {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => {
      const canvas = document.createElement("canvas");
      canvas.width = img.naturalWidth;
      canvas.height = img.naturalHeight;
      const ctx = canvas.getContext("2d");
      if (!ctx) return reject(new Error("no 2d context"));
      ctx.drawImage(img, 0, 0);
      resolve(ctx.getImageData(0, 0, canvas.width, canvas.height));
    };
    img.onerror = reject;
    img.src = url;
  });
}

// Loads a section's relief once per URL pair; null until both images are in (or if either fails).
function useReliefField(relief: string | null, color: string | null): PinField | null {
  const [field, setField] = useState<PinField | null>(null);
  useEffect(() => {
    if (!relief || !color) return setField(null);
    let alive = true;
    Promise.all([loadPixels(relief), loadPixels(color)])
      .then(([h, c]) => alive && setField(fieldFromPixels(h, c)))
      .catch(() => alive && setField(null));
    return () => { alive = false; };
  }, [relief, color]);
  return field;
}

// The 10,000-pin simulation for the current section. Mounted only while open, so the scene lives exactly as long.
export function PinModal({ section, onClose }: { section: Section | null; onClose: () => void }) {
  const [step, setStep] = useState(RESOLUTIONS.length - 1); // start at 100 x 100
  const n = RESOLUTIONS[step];
  const field = useReliefField(section?.relief ?? null, section?.relief_color ?? null);
  const gridKey = section?.grid.flat().join("");
  const shown = useMemo(() => {
    if (!field || !section) return null;
    const f = resample(field, n);
    return n === 10 ? boardField(f, section.grid) : f;
  }, [field, n, gridKey]); // gridKey, not section: polls create a new section object with the same grid

  const hostRef = useRef<HTMLDivElement>(null);
  const sceneRef = useRef<PinScene | null>(null);
  const [noWebgl, setNoWebgl] = useState(false);
  useEffect(() => {
    if (!hostRef.current) return;
    try {
      sceneRef.current = createPinScene(hostRef.current);
    } catch {
      setNoWebgl(true);
    }
    return () => { sceneRef.current?.dispose(); sceneRef.current = null; };
  }, []);
  useEffect(() => { if (shown) sceneRef.current?.show(shown); }, [shown]);

  const message = noWebgl ? COPY.pinsNoWebgl : !section ? COPY.pinsNone : !field ? COPY.pinsMaking : null;

  return (
    <div className="pins" onClick={onClose}>
      <div className="pins__panel" role="dialog" aria-modal="true" aria-labelledby="pins-title" onClick={(e) => e.stopPropagation()}>
        <header className="pins__head">
          <div>
            <span className="label pins__kicker">{COPY.pinsTitle}</span>
            <h2 id="pins-title" className="pins__title">{section?.name ?? COPY.pinsTitle}</h2>
          </div>
          <button type="button" className="key-button label" onClick={onClose}>
            <kbd>P</kbd> {COPY.close}
          </button>
        </header>
        <div className="pins__stage">
          <div ref={hostRef} className="pins__canvas" />
          {message && <p className="pins__message">{message}</p>}
        </div>
        <footer className="pins__foot">
          <label className="pins__slider label">
            {COPY.pinsPerSide}
            <input type="range" min={0} max={RESOLUTIONS.length - 1} step={1} value={step}
                   onChange={(e) => setStep(Number(e.target.value))} />
            <span className="pins__value">{n} × {n}{n === 10 ? ` · ${COPY.pinsToday}` : ""}</span>
          </label>
          <span className="label pins__note">{COPY.pinsKicker} {COPY.pinsHint}</span>
        </footer>
      </div>
    </div>
  );
}
```
